# ── Core LLM / RAG orchestration ──────────────────────────────────────────────
# langchain         : the main orchestration framework; provides chains,
#                     retrievers, and memory abstractions that wire everything
#                     together without boilerplate.
langchain

# langchain-huggingface: official LangChain integration for Hugging Face
#                     models.  Provides HuggingFaceEmbeddings, which wraps
#                     sentence-transformers for local (no-API-key) embedding.
#                     Split from langchain-community in LangChain v0.2 so that
#                     HF-specific updates ship independently of the core.
langchain-huggingface

# sentence-transformers: the underlying library that loads and runs the
#                     all-MiniLM-L6-v2 model (and any other SBERT-family
#                     model).  Downloads model weights from the Hugging Face
#                     Hub on first use and caches them locally (~80 MB for
#                     MiniLM-L6).  No GPU or API key required — inference
#                     runs on CPU out of the box.
sentence-transformers

# langchain-community: third-party loaders, retrievers, and vector-store
#                     wrappers (e.g. Chroma) that the community maintains.
#                     Separating "core" from "community" lets LangChain evolve
#                     the stable API without breaking experimental integrations.
langchain-community

# langchain-text-splitters: text splitting utilities extracted from langchain
#                     core in v0.2.  RecursiveCharacterTextSplitter now lives
#                     here rather than in `langchain.text_splitter`.  Kept as a
#                     separate package so splitter improvements can be shipped
#                     independently of the core orchestration framework.
langchain-text-splitters

# ── Vector store ───────────────────────────────────────────────────────────────
# chromadb          : an open-source, embedded vector database that runs
#                     locally with no external service. Ideal for prototyping
#                     because there is nothing to deploy; swap for Pinecone /
#                     Weaviate when you need cloud scale.
chromadb

# ── Tokenisation ───────────────────────────────────────────────────────────────
# tiktoken          : OpenAI's fast BPE tokeniser. We use it in the chunker to
#                     measure chunk sizes in *tokens* rather than characters,
#                     which is what the model's context window actually counts.
tiktoken

# ── HTTP / data fetching ───────────────────────────────────────────────────────
# requests          : de-facto standard HTTP library; used for any supplemental
#                     REST calls (e.g. fetching market data from a public API).
requests

# ── Configuration ──────────────────────────────────────────────────────────────
# python-dotenv     : loads variables from a .env file into os.environ so that
#                     secrets (API keys) never have to be hard-coded in source.
python-dotenv

# ── SEC filing ingestion ───────────────────────────────────────────────────────
# sec-edgar-downloader: a thin wrapper around the SEC EDGAR full-text search
#                     API. Handles rate limiting, retry logic, and filing
#                     navigation so our downloader.py stays focused on business
#                     logic rather than HTTP plumbing.
sec-edgar-downloader

# ── HTML parsing / cleaning ────────────────────────────────────────────────────
# beautifulsoup4    : HTML/XML parsing library used by cleaner.py to strip all
#                     markup from SEC iXBRL filings.  BeautifulSoup handles
#                     malformed tags, HTML entities, and mixed SGML/HTML
#                     conventions that are common in EDGAR full-submission files.
beautifulsoup4

# lxml              : fast C-based XML/HTML parser used as BeautifulSoup's
#                     backend.  Chosen over the built-in html.parser for its
#                     speed (~5-10× faster) on the large (500 K+ token) filings
#                     we ingest, and for its tolerance of malformed markup.
lxml

# ── LLM provider — Google Gemini ──────────────────────────────────────────────
# langchain-google-genai: LangChain's official integration for Google's
#                     Gemini family of models.  Provides ChatGoogleGenerativeAI,
#                     which wraps the Google GenAI SDK in the standard LangChain
#                     chat-model interface (invoke, stream, batch).  Kept as a
#                     separate package so switching to a different provider
#                     (Anthropic, Mistral) only requires swapping this dependency.
langchain-google-genai

# google-generativeai: the underlying Google GenAI Python SDK that
#                     langchain-google-genai calls under the hood.  Pinned
#                     explicitly so we control the SDK version independently
#                     of LangChain's minimum requirement.
google-generativeai

# ── API layer ──────────────────────────────────────────────────────────────────
# fastapi           : async web framework for exposing the agent as a REST API.
#                     Chosen for its automatic OpenAPI docs (great for demos)
#                     and native async support, which avoids blocking when
#                     awaiting LLM responses.
fastapi

# uvicorn           : ASGI server that runs the FastAPI app. The standard
#                     production-grade choice; supports hot-reload in dev with
#                     --reload flag.
uvicorn

# ── MCP server ─────────────────────────────────────────────────────────────────
# mcp[cli]          : Anthropic's Model Context Protocol Python SDK.  The [cli]
#                     extra adds the `mcp` command-line tool (mcp dev, mcp run)
#                     used for local development and inspection.  The core SDK
#                     provides FastMCP (high-level server API), the Context type
#                     for lifespan-state injection, and transport implementations
#                     (stdio, SSE, streamable-http).
mcp[cli]

# ── MCP client adapter for LangGraph ──────────────────────────────────────────
# langchain-mcp-adapters: bridges MCP servers and LangChain/LangGraph. Provides
#                     MultiServerMCPClient which connects to one or more MCP
#                     servers (stdio, SSE, streamable-HTTP, or WebSocket), runs
#                     the tools/list RPC, and wraps each returned tool in a
#                     LangChain StructuredTool. That lets LangGraph's
#                     create_react_agent consume MCP tools with no code changes
#                     to the reasoning loop — only the tool *source* changes.
langchain-mcp-adapters
